3. Content Analysis Tools:
   ```python
   def analyze_content(content: dict) -> dict:
       """Analyze web content
       Args:
           content: Extracted web content
       Returns:
           Analysis results
       """
       try:
           # 1. Clean text
           text = content.get("text", "").strip()
           if not text:
               return {}
           
           # 2. Summarize text
           processor = TextProcessor()
           summary = processor.summarize(text)
           
           # 3. Extract keywords
           keywords = processor.extract_keywords(text)
           
           # 4. Sentiment analysis
           sentiment = processor.analyze_sentiment(text)
           
           return {
               "title": content.get("title", ""),
               "url": content.get("url", ""),
               "summary": summary,
               "keywords": keywords,
               "sentiment": sentiment
           }
           
       except Exception as e:
           print(f"[ERROR] Content analysis failed: {str(e)}")
           return {}
   ```
